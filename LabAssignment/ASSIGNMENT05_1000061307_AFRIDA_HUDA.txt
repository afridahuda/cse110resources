import java.util.Scanner;
public class TASK1{
    public static void main(String [] args){
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter row: ");
        int row = sc.nextInt();
        int i = 1;
        for ( i = 1; i <= row; i++){
            for (int s = 1; s <= row-i; s++){
                System.out.print(" ");
            }
            for(int j = 1; j <= (2*i)-1; j++){
               System.out.print(j);
            }
            System.out.println();
        }
        for ( i = row-1; i >= 1; i--){
            for (int s = 1; s <= row-i; s++){
                System.out.print(" ");
            }
            for(int j = 1; j <= (2*i)-1; j++){
               System.out.print(j);
            }
            System.out.println();
        }
    }
}

import java.util.Scanner;
public class TASK7{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter row: ");
        int row = sc.nextInt();
        for (int i = row; i >= 1; i -= 2){
            for (int s = 1; s <= (row - i) / 2; s++){
                System.out.print(" ");
            }
            for (int j = 1; j <= i; j++){
                System.out.print(j);
            }
            System.out.println();
        }
        for (int i = 3; i <= row; i += 2){
            for (int s = 1; s <= (row - i) / 2; s++){
                System.out.print(" ");
            }
            for (int j = 1; j <= i; j++){
                System.out.print(j);
            }
            System.out.println();
        }
    }
}
